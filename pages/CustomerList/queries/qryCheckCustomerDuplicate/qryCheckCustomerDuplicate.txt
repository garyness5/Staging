select
    c.id
from public.customers c
where c.client_id = {{ appsmith.store.current_client_id }}::uuid
  and lower(trim(c.customer_name)) =
      lower(trim({{ (inpCustomerName.text || "").trim() }}::text))
  and coalesce(c.deleted, false) = false
  and c.id <> coalesce(
      nullif(
          {{ Number(appsmith.store.current_customer_id || 0) }}::bigint,
          0
      ),
      -1::bigint
  )
limit 1;
