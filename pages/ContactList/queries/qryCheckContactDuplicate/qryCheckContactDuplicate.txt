select id
from public.contacts
where client_id = {{ appsmith.store.current_client_id }}::uuid
  and lower(trim(contact_name)) = lower(trim({{ inpContactName.text }}::text))
  and coalesce(deleted, false) = false
  and (
    {{ !appsmith.store.current_contact_id }}
    or id <> {{ appsmith.store.current_contact_id || 0 }}::bigint
  )
limit 1;
