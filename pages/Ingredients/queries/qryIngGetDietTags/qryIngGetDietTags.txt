select
    hli.id,
    hli.name,
    hli.is_system
from public.helper_list_items hli
join public.helper_lists hl
    on hl.id = hli.list_id
where hli.client_id = '{{ appsmith.store.current_client_id }}'::uuid
  and hl.list_code = 'diet_tag'
  and hli.active = true
  and coalesce(hli.deleted, false) = false
order by
    hli.is_system desc,
    hli.sort_order nulls last,
    hli.name;
