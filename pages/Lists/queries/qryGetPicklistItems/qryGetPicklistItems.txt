select
  id,
  name,
  active,
  is_system
from public.helper_list_items
where list_id = {{
Number(
	this.params.list_id ||
	selListsType.selectedOptionValue ||
	0
)
}}
  and coalesce(deleted, false) = false
  and (
		client_id = '{{ appsmith.store.current_client_id }}'::uuid
		or is_system = true
	)
order by lower(name);
